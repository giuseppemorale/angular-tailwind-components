# Input OTP

`TailwindInputOtp` is a group of single-character fields for OTP / PIN codes, with paste support, keyboard navigation and `ControlValueAccessor` integration.

---

## Import

```typescript
import { TailwindInputOtp } from 'angular-tailwind-components';
```

## Example

<Canvas of={InputOtpStories.OTP} />

```typescript
@Component({
  imports: [TailwindInputOtp],
  template: ` <tailwind-input-otp
    label="Verification code"
    [length]="6"
    [(value)]="code"
    separator="-"
    [separatorAfterIndex]="2"
    (completed)="onComplete($event)" />`
})
export class ExampleComponent {
  code = '';

  onComplete(full: string) {
    console.log('OTP:', full);
  }
}
```
