# Select

`TailwindSelect` is a custom Tailwind dropdown with label, options, keyboard navigation, optional multiple selection and error states (`ControlValueAccessor`).

---

## Import

```typescript
import { TailwindOption, TailwindSelect } from 'angular-tailwind-components';
```

## Properties

<ArgTypes of={SelectStories} />

### `TailwindOption`

```typescript
interface TailwindOption<T = unknown> {
  value: T;
  label: string;
  disabled?: boolean;
}
```

---

## Example

<Canvas of={SelectStories.Select} />

```typescript
@Component({
  imports: [TailwindSelect],
  template: ` <tailwind-select
    label="Framework"
    placeholder="Choose a framework"
    [options]="options"
    size="md"
    [(value)]="value" />`
})
export class ExampleComponent {
  value = '';
  options = [
    { value: 'angular', label: 'Angular' },
    { value: 'react', label: 'React', disabled: true }
  ];
}
```

## Multiple selection

With `[multiple]="true"` the value is a `T[]` array (empty when nothing is selected). The trigger shows the selected entries as removable `tailwind-chip`s; the panel stays open while options are added or removed.

```typescript
@Component({
  imports: [TailwindSelect],
  template: ` <tailwind-select
    label="Framework"
    placeholder="Choose one or more frameworks"
    [options]="options"
    [multiple]="true"
    [(value)]="value" />`
})
export class ExampleComponent {
  value: string[] = [];
  options = [
    { value: 'angular', label: 'Angular' },
    { value: 'react', label: 'React' }
  ];
}
```
